            wb.save(EXCEL_FILE)
            return
    sheet.append([user_id, status])

            break

async def buy(update: Update, context: ContextTypes.DEFAULT_TYPE):
    user = update.effective_user
    payment_uri = f"tron:{WALLET_ADDRESS}?amount={EXPECTED_AMOUNT}"
    msg = (
        f"💸 Send **${EXPECTED_AMOUNT} USDT (TRC20)** to this wallet\n\n"
        f"Public Address:\n`{WALLET_ADDRESS}`\n\n"
        f"Then type /confirm to notify us."
    )

    qr = qrcode.make(payment_uri)
    bio = BytesIO()
    bio.name = "tron_qr.png"
    qr.save(bio, format="PNG")
    bio.seek(0)

    # Save user with more details
    save_new_user(
        user.id,
        user.username or '',
        user.first_name or '',
        user.last_name or '',
        EXPECTED_AMOUNT,
        status='pending'
    )

    await update.message.reply_photo(photo=InputFile(bio), caption=msg, parse_mode="Markdown")

    update_payment_status(user.id, 'pending')
    await update.message.reply_text(

async def handle_screenshot(update: Update, context: ContextTypes.DEFAULT_TYPE):
    user = update.effective_user

    if not update.message.photo:
        await update.message.reply_text("Please send a valid photo.")
        return

    file_id = update.message.photo[-1].file_id
    update_screenshot_file_id(user.id, file_id)

    # Forward screenshot to admin
    await context.bot.forward_message(
        chat_id=ADMIN_ID,  # Your Telegram ID
        from_chat_id=update.message.chat_id,
        message_id=update.message.message_id
    )

    status = None
    for row in sheet.iter_rows(min_row=2):
        if row[0].value == user.id:
            status = row[5].value
            break

    if status == 'paid':
        await update.message.reply_text("✅ Payment confirmed! You already have the download link.")
    elif status == 'pending':
        await update.message.reply_text("📥 Screenshot received! We will verify your payment manually.")
    else:
        await update.message.reply_text("ℹ️ Please use /buy to get payment info first.")


async def mark_paid(update: Update, context: ContextTypes.DEFAULT_TYPE):

        text=f"🎉 Your payment has been verified!\n🔗 Download your product here:\n{PRODUCT_LINK}"

async def admin_mark_paid(update: Update, context: ContextTypes.DEFAULT_TYPE):
    """Admin-only: /markpaid <user_id>"""
    user = update.effective_user
    if user.id != ADMIN_ID:
        await update.message.reply_text("🚫 You are not authorized to use this command.")
        return

    if not context.args:
        await update.message.reply_text("Usage: /markpaid <user_id>")
        return

    try:
        target_user_id = int(context.args[0])
    except ValueError:
        await update.message.reply_text("❌ Invalid user ID.")
        return

    # Update Excel
    update_payment_status(target_user_id, 'paid')

    # Send product to the user
    try:
        await context.bot.send_message(
            chat_id=target_user_id,
            text=f"🎉 Your payment has been verified!\n🔗 Download your product here:\n{PRODUCT_LINK}"
        )
        await update.message.reply_text(f"✅ User {target_user_id} has been marked as paid and product delivered.")
    except Exception as e:
        await update.message.reply_text(f"⚠️ Could not deliver product to user {target_user_id}. Error: {e}")

    app = ApplicationBuilder().token(BOT_TOKEN).build()

    app.add_handler(CommandHandler("confirm", confirm))

    print("✅ Bot starting...")
    await app.run_polling()

        await query.edit_message_caption(

        await context.bot.send_message(

        )
